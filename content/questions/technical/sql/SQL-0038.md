```sql
-- NOT EXISTS
SELECT user_id FROM purchase p1
WHERE p1.product_id = 'A'
  AND NOT EXISTS (
    SELECT 1 FROM purchase p2
    WHERE p2.user_id = p1.user_id AND p2.product_id = 'B'
  );
```
